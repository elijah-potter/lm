Here you go. This code will read in a file and push the float value of each line to the FloatStack:

#include <stdio.h>
#include <stdlib.h>
#include <stddef.h>

typedef struct FloatStack {
     float *arr;
     size_t length;
} FloatStack;

FloatStack *array_init() {
     FloatStack *arr = malloc(sizeof(FloatStack));
     arr -> length = 0;
     return arr;
}

void array_free(FloatStack *arr) {
     if(sizeof(arr -> arr) > 0) free(arr -> arr);
     free(arr);
}

void array_pop(FloatStack *arr) {
     arr -> length--;
     arr -> arr = realloc(arr -> arr, arr -> length * sizeof(float));
}

void array_push(FloatStack *arr, float n) {
     arr -> arr = realloc(arr -> arr, sizeof(arr -> arr) + sizeof(float));
     arr -> arr[arr -> length++] = n;
}

int main() {
     //Initialize dynamic array.
     FloatStack *arr = array_init();
	FILE *file = fopen("data.txt", "r");
     //Buffer where we read in each line. Maximum length of each line is set to 100 here.
	char buffer[100];
	
     if(!file) {
          printf("Please make sure the file exists.\n");
          exit(1);
     }

     while(fgets(buffer, 100, file)) {
          array_push(arr, atof(buffer));
     }

     fclose(file);

     //Print out all elements in the array.
     for(size_t i = 0; i < arr -> length; i++) {
          printf("Array element: %0.2f\n", arr -> arr[i]);
     }

     return 0;

}
