Certainly, here's a basic clock widget using JavaScript. 

$(document).ready(function() {
  const dialog = $('<div></div>')
    .attr('title', 'Clock')
    .appendTo('body')
    .dialog({
      autoOpen: false,
      width: 250,
      height: 100
    });

// get the current time and convert it to a human-readable format
function showTime() {
  const now = new Date();
  const hours = now.getHours();
  const minutes = now.getMinutes().toString().padStart(2, '0');
  const seconds = now.getSeconds().toString().padStart(2, '0');
  const timeString = `${hours}:${minutes}:${seconds}`;
  dialog.text(timeString);
}

// call showTime every 1000 milliseconds
 setInterval(showTime, 1000);
 
// add a button to show the clock 
const button = $('<button></button>')
    .text('Show Clock')
    .on('click', function() {
      dialog.dialog('open');
    })
    .appendTo('body');
});
 
This code uses JQuery UI's Dialog module along with a JavaScript Date object to display the current time in a small window.  Note that you'll have to include links to the JQuery and JQuery UI libraries in the head of your HTML file like so:

<script>https://code.jquery.com/jquery-3.6.3.js</script>
<script> https://code.jquery.com/ui/1.13.2/jquery-ui.js</script>
