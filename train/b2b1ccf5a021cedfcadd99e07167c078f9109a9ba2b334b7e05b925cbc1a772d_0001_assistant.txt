There are a few things that need to be changed:

- The velocity of a free-falling object increases by 9.8 meters/second every second. You're multiplying the velocity by 9.8 each time dropBall() is called, instead of adding 9.8. 
- dropBall() is called every time the user's monitor refreshes, which is usually 30-60 times a second. That means the box will fall 30-60 times faster than in real life. To avoid this, you can use the setTimeout() function to make sure the ball's position updates after exactly one second has passed. 

Here's a version of your code with these changes implemented:

```Javascript
    <script>
        class FallingCanvas {
            constructor(canvas_id, side_length) {
                this.canvas = document.getElementById(canvas_id);
                this.canvas.addEventListener('click', this.dropBox, false);

                this.velocity = 0;
                this.acceleration = 9.8;
                this.side_length = side_length;
                this.box_y = this.canvas.height * 0;
                this.box_x = (this.canvas.width - this.side_length) / 2;

                this.drawBackground();
                this.drawBox();
            }

            drawBackground = () => {
                const ctx = this.canvas.getContext("2d");
                ctx.fillStyle = "LightSkyBlue";
                ctx.fillRect(0, 0, this.canvas.width, this.canvas.height);
            }

            drawBox = () => {
                const ctx = this.canvas.getContext("2d");
                ctx.fillStyle = 'purple';
                ctx.fillRect(this.box_x, this.box_y, this.side_length, this.side_length);
            }

            dropBox = () => {
                if (this.box_y + this.side_length > this.canvas.height) { return; }
                
                this.box_y += this.velocity;
                this.velocity += this.acceleration;

                this.drawBackground();
                this.drawBox();

                setInterval(this.dropBox, 1000);
            }
        }

        fc = new FallingCanvas('falling_canvas', 50);
    </script>
```
