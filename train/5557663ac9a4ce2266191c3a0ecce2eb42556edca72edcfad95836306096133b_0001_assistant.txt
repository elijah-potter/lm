public class Computer {
    private String computerModel;
    private String CPUModel;
    private double clockSpeed;
    private int RAMSizeInGB;
    private int RAMSpeedInMHz;
    private int SSDSizeInGB;

    public Computer(String computerModel, String CPUModel, double clockSpeed, int RAMSizeInGB, int RAMSpeedInMHz, int SSDSizeInGB) {
        this.computerModel = computerModel;
        this.CPUModel = CPUModel;
        this.clockSpeed = clockSpeed;
        this.RAMSizeInGB = RAMSizeInGB;
        this.RAMSpeedInMHz = RAMSpeedInMHz;
        this.SSDSizeInGB = SSDSizeInGB;
    }

    public String getComputerModel() {
        return this.computerModel;
    }

    public void setComputerModel(String computerModel) {
        this.computerModel = computerModel;
    }

    public String getCPUModel() {
        return this.CPUModel;
    }

    public void setCPUModel(String CPUModel) {
        this.CPUModel = CPUModel;
    }

    public double getClockSpeed() {
        return this.clockSpeed;
    }

    public void setClockSpeed(double clockSpeed) {
        this.clockSpeed = clockSpeed;
    }

    public int getRAMSize() {
        return this.RAMSizeInGB;
    }

    public void setRAMSize(int RAMSizeInGB) {
        this.RAMSizeInGB = RAMSizeInGB;
    }

    public int getRAMSpeed() {
        return this.RAMSpeedInMHz;
    }

    public void setRAMSpeed(int RAMSpeedInMHz) {
        this.RAMSpeedInMHz = RAMSpeedInMHz;
    }

    public int getSSDSize() {
        return this.SSDSizeInGB;
    }

    public void setSSDSize(int SSDSizeInGB) {
        this.SSDSizeInGB = SSDSizeInGB;
    }

    public void printSpecs() {
        System.out.printf(
            "Computer: %s\nCPU: %s, %.1f GHz\nRAM: %d GB, %d MHz\nStorage:%d GB\n",
            this.computerModel, this.CPUModel, this.clockSpeed, this.RAMSizeInGB, this.RAMSpeedInMHz,
            this.SSDSizeInGB
        );
    }
}
