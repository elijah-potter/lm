#Get the Flask class
from flask import Flask
#Declare our app by creating an instance of the class. We pass it the name parameter so that it can reference this file automatically, even if we change its name. 
app = Flask(__name__)

#We use this decorator to select what route brings us to this function, and we pass it in the methods allowed. In this case, only get requests. 
@app.route('/', methods = ['GET'])
def main():
     #Returning a string well send it to the client.
     return 'Hello, world!'

#Run the app only if we're running this file, i.e. if it's not part of a package we imported.
if(__name__ == '__main__'):
     app.run(port = 3000)
