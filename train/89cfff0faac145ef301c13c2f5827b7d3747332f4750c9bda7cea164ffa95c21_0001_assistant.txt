Here is a Python script that generates a random string of letters between 4 and 8 characters in length.

import random
import string

# minimum and maximum password length
min_length = 4
max_length = 8

# define a random length between min and max
length = random.randint(min_length, max_length)

# define and stringify the alphabet
characters = string.ascii_letters

# randomly select letters from the alphabet to create the password
password = ''.join(random.choice(characters) for i in range(length))

# print the password
print(password)
