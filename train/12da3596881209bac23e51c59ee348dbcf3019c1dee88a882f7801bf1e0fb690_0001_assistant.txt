def calculate(string):
    num = [] # use a list to store the numbers
    for word in string.split():
        if word == 'loses':
            operator = '-'
        elif word == 'gains':
            operator = '+'
        elif word.isdigit():
            num.append(word) #append instead of adding
    return eval(str(num[0]) + operator + str(num[1])) #change the return statement to use the list
