function multiGreet(name) {
  const multiHello = [
    "Hello",
    "Hola",
    "������������������",
    "���������������������������������������������",
    "���������������������������������������������",

  ];

console.log(multiHello[Math.floor(Math.random()*multiHello.length)] + ', ' + name + ".")
}

For example, the call multiGreet('Sally') could return "Hello, Sally." or "Hola, Sally."
