Sure thing! here is code that connects the Nicla Sense Me to a device via BLE:

#include "Arduino_BHY2.h"

// initialize object for quaternion data 
SensorQuaternion rotation(SENSOR_ID_RV);


// frequency that the imu data is polled (in HZ)
int poll_time = 1;

void setup(){
  Serial.begin(115200);
  // send IMU data via BLE
  BHY2.begin();
  rotation.begin();
}

void loop(){
  static auto lastCheck  = millis();
  BHY2.update();

  // poll IMU data and print to serial monitor
  if(millis() - lastCheck >= 1000/poll_time) 
  {
    lastCheck = millis();
    Serial.println(String("Device 1: rotation: ") + rotation.toString());
  }
}

Here we can see that the code uses the "BHY2Host.h" library to read the quaternion data from the IMU as well as advertise a service using BLE.
