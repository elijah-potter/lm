Utils.readJson = function (file, callback) {
    // Create an XMLHttpRequest.
    var req = new XMLHttpRequest();
    // Handle errors that may occur when executing the request.
    req.onerror = function (e) {
        // Utils.log(`[Utils][readJson] Error while trying to read  ${file}`, e);
    };
    // Set the MIME type of the request to accept JSON.
    req.overrideMimeType('application/json');
    // Send a GET request to the local JSON file to read it.
    req.open('GET', file, true);
    // Listen for when the file has been read.
    req.onreadystatechange = function () {
        // Checks that the request is done without errors.
        if (req.readyState === 4) {
            // && req.status == "200") {
            // Executes the callback with the received JSON.
            if (callback) callback(req.responseText);
        }
    };
    // Terminates the request by sending null.
    req.send(null);
};
