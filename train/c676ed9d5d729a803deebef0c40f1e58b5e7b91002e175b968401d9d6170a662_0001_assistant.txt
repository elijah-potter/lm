Below is an example of a method handler that allows for the updating of the "title" variable.

const app = Vue.createApp({
  data() {
    return {
      title: 'The Way of Kings',
      author: 'Brandon Sanderson',
      age: 45
    }
  },
  methods: {
    changeTitle(title) {
      // this.title = 'Words of Radiance'
      this.title = title
    }
  }
})

app.mount('#app')
