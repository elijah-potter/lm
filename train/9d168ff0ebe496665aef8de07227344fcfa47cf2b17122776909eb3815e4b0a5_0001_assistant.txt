// Import the Scanner utility to enable the creation of a Scanner object for user inputs
import java.util.Scanner;

/* Define the class, Average, which accepts five integer inputs from the user and
    then calculates the average value. */
public class Average {

    public static void main(String[] args) {

        // Create a new Scanner object named 'in' to read user inputs
        Scanner in = new Scanner(System.in);

        // Prompt and read the first integer from the user
        System.out.print("Input first number: ");
        int num1 = in.nextInt();

        // Prompt and read the second integer from the user
        System.out.print("Input second number: ");
        int num2 = in.nextInt();

        // Prompt and read the third integer from the user
        System.out.print("Input third number: ");
        int num3 = in.nextInt();

        // Prompt and read the fourth integer from the user
        System.out.print("Input fourth number: ");
        int num4 = in.nextInt();

        // Prompt and read the fifth integer from the user
        System.out.print("Enter fifth number: ");
        int num5 = in.nextInt();


        // Calculate the average of the five integers and print the result
        System.out.println("Average of five numbers is: " +
                (num1 + num2 + num3 + num4 + num5) / 5);
    }
}
