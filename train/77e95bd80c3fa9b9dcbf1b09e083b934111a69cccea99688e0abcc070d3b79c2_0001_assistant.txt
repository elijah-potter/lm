# Repeat infinitely.
while true; do
  # Set the variable cpu_usage to the process' CPU usage.
  cpu_usage=$(ps -o %cpu -p $(pgrep $process_name) --no-headers)
  # Check if the CPU usage is greater than the CPU threshold.
  if [[ $(echo "$cpu_usage>$cpu_threshold" | bc) -eq 1 ]]; then
    # Send a notification titled "CPU Alert" to inform the user of the process' high CPU usage.
    notify-send "CPU Alert" "$process_name is using more than $cpu_threshold% CPU"
  fi
  # Wait 1 minute before repeating.
  sleep 1m
done
