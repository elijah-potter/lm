#include <stdio.h>
#include <stdlib.h>
#include <stddef.h>

typedef struct DynamicFloatArray {
     float *arr;
     size_t length;
} DynamicFloatArray;

DynamicFloatArray *array_init() {
     DynamicFloatArray *arr = malloc(sizeof(DynamicFloatArray));
     arr -> length = 0;
     return arr;
}

void array_free(DynamicFloatArray *arr) {
     if(sizeof(arr -> arr) > 0) free(arr -> arr);
     free(arr);
}

void array_push(DynamicFloatArray *arr, float n) {
     arr -> arr = realloc(arr -> arr, sizeof(arr -> arr) + sizeof(float));
     arr -> arr[arr -> length++] = n;
}

int main() {
     //Initialize dynamic array.
     DynamicFloatArray *arr = array_init();

     //Push sample values.
     array_push(arr, 3.72);
      array_push(arr, -4.42);
      array_push(arr, -7.8);
      array_push(arr, 32.44);
      array_push(arr, 8.65);

     //Print out all elements in the array.
     for(size_t i = 0; i < arr -> length; i++) {
          printf("%f\n", arr -> arr[i]);
     }

     return 0;

}
