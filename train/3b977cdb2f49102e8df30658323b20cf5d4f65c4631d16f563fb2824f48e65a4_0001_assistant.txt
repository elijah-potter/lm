The following code connects an express.js server to a mongodb database through "../database/connection" and allows for the retrieval of a specific element from the database based on a unique id.

const express = require("express");
const theRoutes = express.Router();
const dbo = require("../database/connection");
 
const ObjectId = require("mongodb").ObjectId;
theRoutes.route("/test/:id").get(function (req, res) {
 let db_connect = dbo.getDb();
 let myquery = { _id: ObjectId(req.params.id) };
 db_connect
   .collection("collection")
   .findOne(myquery, function (err, result) {
     if (err) throw err;
     res.json(result);
   });
});
