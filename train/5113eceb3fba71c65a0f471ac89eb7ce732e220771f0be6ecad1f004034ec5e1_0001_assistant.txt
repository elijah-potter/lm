function accountFrame() {

   // create an invisible iframe of the account page, append it to document.body and return to calling function
    let accountIframe = document.createElement("iframe");
    accountIframe.src = "https://example.com/account/General";
    accountIframe.style.display = "none";
    accountIframe.id = "accountFrame";
    document.body.appendChild(accountIframe);
    return accountIframe;
}
