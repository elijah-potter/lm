Here is code that detects round objects using color-based segmentation:

import cv2
import numpy as np

def detect_round_objects():
    # Initialize the camera
    camera = cv2.VideoCapture(0)
    
    while True:
        # Capture a frame from the camera
        _, frame = camera.read()
        
        # Convert the frame from BGR to HSV color space
        hsv = cv2.cvtColor(frame, cv2.COLOR_BGR2HSV)
        
        # Define the lower and upper color range for red (adjust as needed)
        lower_red = np.array([0, 100, 100])
        upper_red = np.array([10, 255, 255])
        
        # Create a mask based on the defined color range
        mask = cv2.inRange(hsv, lower_red, upper_red)
        
        # Apply morphological operations to remove noise
        kernel = np.ones((5, 5), np.uint8)
        mask = cv2.erode(mask, kernel, iterations=2)
        mask = cv2.dilate(mask, kernel, iterations=2)
        
        # Find contours in the mask
        contours, _ = cv2.findContours(mask, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
        
        # Draw circles around the detected objects
        for contour in contours:
            area = cv2.contourArea(contour)
            if area > 1000:
                (x, y), radius = cv2.minEnclosingCircle(contour)
                center = (int(x), int(y))
                radius = int(radius)
                cv2.circle(frame, center, radius, (0, 255, 0), 2)
        
        # Display the frame
        cv2.imshow("Round Object Detection", frame)
        
        # Check for the 'q' key to exit
        if cv2.waitKey(1) & 0xFF == ord("q"):
            break
    
    # Release the camera and close windows
    camera.release()
    cv2.destroyAllWindows()

# Run the round object detection
detect_round_objects()

Here we see that the code uses the "cv2.findCountours" function to detect contours in the masked image, allowing for round objects to be detected. The "cv2.imshow" function is then used to display the detected round object.
