Below is an example of the v-for directive being used to loop through items in an array:

<ul id="list">
  <li v-for="(stuff, index) in items">
    {{ index }} - {{ stuff.message }}
  </li>
</ul>

var vm = new Vue({
  el: '#list',
  data: {
    stuff: [
      { message: 'Michael' },
      { message: 'Cera' }
    ]
  }
})
