Here is an example of a program that solves for the first number that can be written as the sum of primes in over 3 thousand different ways.

import eulerlib, itertools

def sumOfPrimes():
	cond = lambda n: num_prime_sum_ways(n) >= 3000
	ans = next(filter(cond, itertools.count(2)))
	return str(ans)
