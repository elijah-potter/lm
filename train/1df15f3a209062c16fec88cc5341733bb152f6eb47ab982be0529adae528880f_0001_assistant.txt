You can use CSS modules to use conventional CSS stylesheets with your React Native application. 
Here is the code:

import React from 'react';
import { View, Text, SafeAreaView } from 'react-native';
import style from './App.module.css'; // CSS file imported 

export default function App() {
  return (
    <>
      <SafeAreaView>
        <View style={style.container}>
          <Text style={style.text}>Alpha</Text>
          <Text style={style.text}>Delta</Text>
        </View>
      </SafeAreaView>
    </>
  );
}

with the Css file looking something like this:

.container {
  flex-flow: column;
  justify-content: space-evenly;
  align-items: center;
  width: '100%';
  height: 20;
}

.text {
  color: '#ffffff';
}
