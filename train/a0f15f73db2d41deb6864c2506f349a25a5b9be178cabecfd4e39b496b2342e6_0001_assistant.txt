Here is a short Python function that will create a random password based on your given parameters.

import string
import random

def generate_password():
    # Define the character sets for each category
    special_chars = string.punctuation
    lowercase_chars = string.ascii_lowercase
    uppercase_chars = string.ascii_uppercase
    numeric_chars = string.digits

    # Generate a password by randomly selecting characters from each category
    password = random.choice(special_chars)
    password += random.choice(lowercase_chars)
    password += random.choice(uppercase_chars)
    password += random.choice(numeric_chars)
    
    return password
