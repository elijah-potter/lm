import socket

def tcp_connection(host, port):
    # Create a TCP socket
    sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    
    try:
        # Connect to the server
        sock.connect((host, port))
        print(f"Connected to {host}:{port}")
        
        while True:
            # Receive data from the server
            data = sock.recv(1024)
            
            if not data:
                # No more data to receive
                break
            
            # Decode the TCP packet content
            packet_content = data.decode()
            
            # Print the formatted packet content
            print("Received packet:")
            print("-" * 20)
            print(packet_content)
            print("-" * 20)
    
    except ConnectionRefusedError:
        print(f"Connection to {host}:{port} refused")
    
    except Exception as e:
        print(f"An error occurred: {str(e)}")
    
    finally:
        # Close the socket
        sock.close()

# Usage example
tcp_connection("example.com", 80)

Here we can see that the code creates a tcp client socket and attempts to connect to a host (example.com) on port 80. The code then reads the packet data using "data.decode". Finally, the tcp packet data is printed in a readable format.
